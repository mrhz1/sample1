import cmlapi

def fields(cls):
    return getattr(cls, "openapi_types", None) or getattr(cls, "swagger_types", None)

# 1. All request/model classes related to teams or quotas
print("== Models with Team/Quota in the name ==")
for n in sorted(dir(cmlapi)):
    if ("Team" in n or "Quota" in n) and isinstance(getattr(cmlapi, n), type):
        print(" ", n)

# 2. Fields of the key models
for name in ["CreateTeamRequest", "CreateProjectRequest",
             "UpdateAcceleratorBasedTeamQuotaRequest"]:
    cls = getattr(cmlapi, name, None)
    print(f"\n== {name} ==")
    print(fields(cls) if cls else "not found")

# 3. Method docs
for m in ["create_team", "update_accelerator_based_team_quota",
          "set_team_default_quota", "delete_team"]:
    doc = (getattr(cmlapi.CMLServiceApi, m).__doc__ or "").strip().splitlines()
    print(f"\n== {m} ==")
    print("\n".join(doc[:12]))
