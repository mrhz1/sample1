$out = "C:\Temp\controllers.txt"
Get-ChildItem src -Recurse -Include *Controller.cs | Where-Object { $_.FullName -notmatch '\\(bin|obj)\\' } | ForEach-Object {
  "== $($_.Name)"
  Select-String -Path $_.FullName -Pattern 'DomainAuthorize|\[Http(Post|Get)\]|public .*Result\s+\w+\(' | ForEach-Object { "   " + $_.Line.Trim() }
} | Set-Content $out
Get-ChildItem src -Directory -Recurse | Where-Object { $_.FullName -notmatch '\\(bin|obj|packages)\\?' } | ForEach-Object { $_.FullName.Replace((Get-Location).Path + '\','') } | Add-Content $out
code $out
