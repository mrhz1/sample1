def try_move(target):
    try:
        api.update_project_file_metadata(cmlapi.FileInfo(path=target), PID, "hive_probe.txt")
        folder = target.rsplit("/", 1)[0]
        print("MOVED to", target, "->", [f.path for f in api.list_project_files(PID, folder).files])
        return True
    except Exception as e:
        print("MOVE to", target, "FAILED:", type(e).__name__, getattr(e, "status", ""), getattr(e, "body", e))
        return False
