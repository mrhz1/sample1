import cmlapi, os, getpass

print("Running as OS user:", getpass.getuser())
print("Hadoop user:", os.environ.get("HADOOP_USER_NAME"))

c = cmlapi.default_client()

def tryit(label, fn):
    try:
        fn(); print(f"[OK]   {label}")
    except cmlapi.rest.ApiException as e:
        print(f"[FAIL] {label}: {e.status} {e.body[:120]}")

tryit("list projects", lambda: c.list_projects(page_size=1))
tryit("admin check (default quotas)", lambda: c.get_default_quotas())

p = c.list_projects(page_size=5).projects
print("Projects visible:", [x.name for x in p])
